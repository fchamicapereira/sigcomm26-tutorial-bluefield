% each guide's own front matter (title, subtitle). See README.md.

% Built with xelatex, for the Unicode in the schedule line (·, –) and in institution names.

% Pandoc's own preamble, pulled in rather than hand-copied. It brings whatever the document
% actually uses -- tables (longtable/booktabs/calc, and the \newcounter{none} that an uncaptioned
% table steps), syntax-highlighting macros, \tightlist, image scaling, strikeout -- each behind a
% conditional on a feature pandoc detected in the source. Reimplementing this by hand is how a
% template ends up broken by a pandoc upgrade: the first table added to a guide failed with
% "No counter 'none' defined" before this line existed.
%
% NB: pandoc parses this whole file as a template and knows nothing about LaTeX comments, so a
% literal dollar-if in the prose above would be read as a directive and fail the build. Keep
% template syntax out of the comments.
$common.latex()$

% After common.latex, so hyperref loads last as it expects to.
%
% Deliberately NOT [hidelinks]. Every link in these guides is a download page or a fallback package
% the reader is meant to click, and hidden links are typeset as ordinary prose -- indistinguishable
% from the text around them, so nothing tells the reader they are there at all.
%
% Coloured rather than framed: hyperref's default is a box around each link, which is fine on screen
% and prints as a stray rectangle. The two colours are acmart's own -- ACMDarkBlue for external URLs
% and ACMPurple for internal references -- so a guide read next to the SIGCOMM papers matches them.
% Both are defined here because acmart itself is not loaded (see the font stack below, which copies
% from it for the same reason).
\usepackage{hyperref}
\definecolor{ACMDarkBlue}{cmyk}{1,0.58,0,0.21}
\definecolor{ACMPurple}{cmyk}{0.55,1,0,0.15}
\hypersetup{
  colorlinks = true,
  urlcolor   = ACMDarkBlue,
  linkcolor  = ACMPurple,
  citecolor  = ACMPurple,
}

\geometry{margin=2.4cm,top=2.0cm,bottom=2.2cm}

% Pandoc's own default template sets this; ours replaces that template wholesale and so dropped it.
% Restored here on its own merits -- it gives TeX a last-resort pass with extra stretchability, which
% keeps ordinary prose from spilling into the margin.
%
% Do NOT expect it to fix the overfull lines these guides actually hit. Measured: the seven in
% doca-flow.md were unchanged, to the point, with and without it. Those come from long unbreakable
% \texttt tokens -- 30-to-40-character DOCA identifiers, absolute paths -- which have no hyphenation
% points at all, so TeX has no better break to find however much slack it is given. The fix for
% those is in the Markdown: shorten the token, or reword so it does not land mid-sentence.
%
% `make check` finds them. Pandoc does not report overfull boxes at all, so a plain build cannot.
\setlength{\emergencystretch}{3em}

% No section numbers. Guides name their sections "Step 1 -- ...", and LaTeX's own numbering on top
% of that produces "2 Step 1 -- ...", which is worse than useless in a document people follow in
% order while typing.
\setcounter{secnumdepth}{0}

% Heading fonts copied from acmart's sigconf format -- the one SIGCOMM uses:
%
%   \def\@secfont{\bfseries\Large\section@raggedright}      bold SERIF, not sans
%   \def\@subsecfont{\bfseries\Large\section@raggedright}
%   \def\@subsubsecfont{\sffamily\itshape}                  sans italic
%
% Worth stating plainly because it is the opposite of what the sans title block suggests: only
% manuscript/acmsmall set headings in sans. sigconf sets them in bold Libertine.
%
% One deliberate deviation. acmart gives \section and \subsection the SAME size and tells them
% apart by their numbers; these guides run with secnumdepth=0 (see below), so equal sizes would
% leave the two levels indistinguishable. \subsection therefore drops to \large.
%
% \raggedright matches acmart's \section@raggedright. It earns its place here: guide headings are
% written as full sentences ("Step 1 -- Install the client", and longer), so they wrap, and a
% justified wrapped heading comes out with stretched word spaces.
\titleformat{\section}{\bfseries\Large\raggedright}{}{0pt}{}
\titleformat{\subsection}{\bfseries\large\raggedright}{}{0pt}{}
\titleformat{\subsubsection}{\sffamily\itshape\normalsize\raggedright}{}{0pt}{}
\titlespacing*{\section}{0pt}{18pt}{6pt}
\titlespacing*{\subsection}{0pt}{13pt}{4pt}
\titlespacing*{\subsubsection}{0pt}{11pt}{3pt}

\setlength{\parindent}{0pt}
\setlength{\parskip}{6pt}

% ACM's font stack, as the acmart class itself sets it:
%
%   \RequirePackage[tt=false, type1=true]{libertine}   Linux Libertine (serif) + Linux Biolinum (sans)
%   \RequirePackage[varqu]{zi4}                        Inconsolata (mono), with upright quotes
%
% All three ship with TeX Live, so they need no font installation and no per-platform
% substitution. Math is deliberately left alone: acmart adds newtxmath, but no guide contains any,
% and loading it under xelatex/fontspec buys nothing here.
%
% Addressed by FILE name rather than family name ("Linux Libertine O") on purpose: TeX Live does
% not register its fonts with the operating system, so on a machine whose fontconfig/CoreText has
% never heard of them -- macOS, out of the box -- a family-name lookup fails and the build dies at
% \setmainfont. Filenames resolve through kpathsea, which always knows where TeX Live put them.
\setmainfont{LinLibertine}[
  Extension      = .otf,
  UprightFont    = *_R,
  BoldFont       = *_RB,
  ItalicFont     = *_RI,
  BoldItalicFont = *_RBI,
]
% Biolinum has no true bold italic; _RBO is its bold OBLIQUE, which is what the libertine package
% substitutes there too.
\setsansfont{LinBiolinum}[
  Extension      = .otf,
  UprightFont    = *_R,
  BoldFont       = *_RB,
  ItalicFont     = *_RI,
  BoldItalicFont = *_RBO,
]
% Inconsolata is the one piece of this stack that a working xelatex does not already guarantee.
% TeX Live ships libertine in BOTH collection-fontsextra and collection-xetex, so any install able
% to run this template has Linux Libertine and Biolinum; inconsolata is in collection-fontsextra
% only. Say so, rather than letting fontspec fail with a bare "font cannot be found" that gives no
% hint which package supplies it.
\IfFontExistsTF{Inconsolatazi4-Regular.otf}{}{%
  \errmessage{Inconsolata is missing -- run: tlmgr install inconsolata (it is ACM's monospace font,
    and unlike the rest of this stack it is not part of collection-xetex)}%
}
% Inconsolata has NO italic at all -- its own inconsolata.fontspec declares only upright and bold,
% and zi4's .fd files substitute the upright shape for it. Both italic slots are therefore mapped
% explicitly to the real faces: that is what ACM effectively renders, and it keeps the build free
% of "font shape undefined" warnings on every code comment. Tango sets comments in italic, so they
% come out upright here and are told apart by colour alone, exactly as in an ACM paper.
%
% StylisticSet 3 is the OpenType equivalent of zi4's varqu option: it swaps in upright quotesingle,
% quotedbl and grave. That matters in a document that is mostly shell commands -- a curly quote in
% a command someone retypes does not work.
\setmonofont{Inconsolatazi4}[
  Extension      = .otf,
  UprightFont    = *-Regular,
  BoldFont       = *-Bold,
  ItalicFont     = *-Regular,
  BoldItalicFont = *-Bold,
  StylisticSet   = 3,
  Scale          = MatchLowercase,
]

% Box-drawing, arrow and shape glyphs, which Inconsolata does not have a single one of.
%
% This matters more than it looks. xelatex DROPS a character its font cannot render, warns only on
% stderr, and still exits 0 -- so an ASCII diagram in a fenced code block loses every one of its
% lines and the PDF builds "successfully" with a hole where the figure was. `make check` is what
% catches it; this is what fixes it. (Measured: guides/doca-pcc.md dropped 199 glyphs from its two
% ASCII diagrams before this block existed.)
%
% XeLaTeX has no automatic font fallback -- that is a LuaLaTeX feature -- so every character has to
% be mapped by hand. DejaVu Sans Mono has all of them and ships with TeX Live in the same
% collection-fontsextra that already supplies Inconsolata, so this adds no new install requirement.
%
% newunicodechar makes each character active, and that DOES reach inside fancyvrb's Verbatim, which
% is where pandoc puts fenced code blocks: fancyvrb rewrites catcodes for the ASCII specials but
% leaves everything above 127 alone. Verified on a real code block, not assumed.
%
% The scale is chosen to match ADVANCE WIDTHS, not x-heights, and that is the whole trick for ASCII
% art. DejaVu Sans Mono's cell is 0.6021em, Inconsolata's is 0.5000em, so an unscaled fallback runs
% 20% wide: box-drawing verticals drift right of the column they belong to, and a 90-column diagram
% stops fitting the text block. 0.5000/0.6021 = 0.8305 puts one fallback glyph in exactly one
% Inconsolata cell, so the grid holds and the art lines up. Scale=MatchLowercase would look tidier
% on a mixed prose line and wreck every diagram; diagrams are the reason this block exists.
\usepackage{newunicodechar}
\newfontfamily\fallbackmono{DejaVuSansMono}[
  Extension   = .ttf,
  UprightFont = *,
  BoldFont    = *-Bold,
  Scale       = 0.8305,
]
\newcommand{\fbmono}[1]{{\fallbackmono #1}}
% Listed explicitly rather than looped, so grepping for a character that came out wrong finds it.
\newunicodechar{─}{\fbmono{─}}  \newunicodechar{│}{\fbmono{│}}
\newunicodechar{┌}{\fbmono{┌}}  \newunicodechar{┐}{\fbmono{┐}}
\newunicodechar{└}{\fbmono{└}}  \newunicodechar{┘}{\fbmono{┘}}

\newunicodechar{┼}{\fbmono{┼}}

\newunicodechar{╔}{\fbmono{╔}}  \newunicodechar{╗}{\fbmono{╗}}
\newunicodechar{╚}{\fbmono{╚}}  \newunicodechar{╝}{\fbmono{╝}}

% --- institution marks -----------------------------------------------------------------------
% Logo files are dropped into logos/ by hand and are not in the repository (see logos/README.md),
% so this must degrade rather than fail: a missing file renders a framed placeholder naming the
% file that is wanted, which is visible in the PDF and obvious in review. Guides therefore build
% correctly from a fresh clone and simply get better as logos land.
\newcommand{\logodir}{$if(logodir)$$logodir$$else$logos$endif$}
\newcommand{\logoheight}{2.1ex}

\newcommand{\logoimg}[1]{\raisebox{-0.35ex}{\includegraphics[height=\logoheight]{#1}}}

% Without a logo file, fall back to the traditional footnote symbols in their conventional order,
% one per institution -- the same marks a paper would use. Each institution is registered once,
% in the order it appears in tutorial.json, so the symbol is a property of the institution and
% every person affiliated with it gets the same one.
\newcounter{institution}
\newcommand{\registerinstitution}[1]{%
  \stepcounter{institution}%
  \expandafter\xdef\csname instnum@#1\endcsname{\number\value{institution}}%
}
% Text-mode asterisks, not math mode: pandoc parses this file as a template, so a dollar-delimited
% math group anywhere in it is read as a directive and fails the build.
\newcommand{\affilsymbol}[1]{%
  \ifcase#1\or *\or \textdagger\or \textdaggerdbl\or \S\or \P\or \textbardbl\or
  **\or \textdagger\textdagger\or \textdaggerdbl\textdaggerdbl\else ?\fi
}
\newcommand{\affilmark}[1]{%
  \IfFileExists{\logodir/#1.pdf}{\,\logoimg{\logodir/#1.pdf}}{%
    \IfFileExists{\logodir/#1.png}{\,\logoimg{\logodir/#1.png}}{%
      \ifcsname instnum@#1\endcsname
        \textsuperscript{\affilsymbol{\csname instnum@#1\endcsname}}%
      \else
        \textsuperscript{?}%
      \fi}}%
}

% Assign the symbols up front, in the preamble, because the author rows are typeset before the
% legend that lists the institutions and both have to agree on which mark belongs to whom.
$for(institutions)$\registerinstitution{$institutions.id$}$endfor$

% --- header block ----------------------------------------------------------------------------
% The rules above and below the header are one command, not two similar-looking literals, so they
% cannot drift apart in weight or colour the next time either end is adjusted.
\newcommand{\headerrule}{\rule{\textwidth}{2pt}\par}
% Author names in sans, matching acmart's \@authorfont (\Large\sffamily for sigconf). Sans is
% reserved throughout this template for the two things acmart uses it for -- the title and the
% names -- and everything else is Libertine.
\newcommand{\peoplerow}[1]{{\sffamily\small #1\par}}

% Colour in the footer must be set with \textcolor, or inside an explicit group -- never with a
% bare \color switch. A bare switch is not bracketed, so it survives the page shipout and the NEXT
% page's body text comes out in it: page 1 reads black (its body is typeset before its footer) and
% everything from page 2 on is grey. Same reason \footrule is wrapped in braces.
\fancypagestyle{guide}{%
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0.4pt}
  \renewcommand{\footrule}{{\color{rule}\hrule height 0.4pt}}
  \fancyfoot[L]{\textcolor{muted}{\footnotesize $conference$ · $title$}}
  \fancyfoot[R]{\textcolor{muted}{\footnotesize \thepage}}
}
\pagestyle{guide}

% Fenced code blocks as a soft-grey panel rather than bare verbatim: these guides are almost
% entirely commands to copy, so the blocks are the content and should read as such. This has to
% come after common.latex, which is where Shaded and Highlighting are defined -- and inside the
% same conditional, because common.latex only defines them for a document that actually contains
% highlighted code. Renewing them unconditionally breaks any guide made purely of prose or tables.
$if(highlighting-macros)$
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},fontsize=\small,breaklines,breakanywhere}
\renewenvironment{Shaded}
  {\begin{tcolorbox}[breakable,colback=shell,colframe=rule,boxrule=0.4pt,arc=2pt,
                     left=6pt,right=6pt,top=5pt,bottom=5pt,enhanced jigsaw]}
  {\end{tcolorbox}}
$endif$
