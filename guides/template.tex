% Shared header for every participant guide in this directory. This is the only LaTeX in the
% repository: guides are written in Markdown and none of them should ever need to touch this file.
%
% Fed by tutorial.json (the shared fields: tutorial title, schedule, people, institutions) and by
% each guide's own front matter (title, subtitle). See README.md.
%
% Built with xelatex, for the Unicode in the schedule line (·, –) and in institution names.
\documentclass[11pt,a4paper]{article}

\usepackage{fontspec}
\usepackage{geometry}
% graphicx unconditionally, unlike the rest: \affilmark below draws logos even in a guide whose
% own Markdown contains no images, and common.latex only loads it when the document has some.
\usepackage{graphicx}
\usepackage{xcolor}
\usepackage{fancyhdr}
\usepackage{enumitem}
\usepackage{fvextra}
\usepackage[most]{tcolorbox}
\usepackage{titlesec}

% Pandoc's own preamble, pulled in rather than hand-copied. It brings whatever the document
% actually uses -- tables (longtable/booktabs/calc, and the \newcounter{none} that an uncaptioned
% table steps), syntax-highlighting macros, \tightlist, image scaling, strikeout -- each behind a
% conditional on a feature pandoc detected in the source. Reimplementing this by hand is how a
% template ends up broken by a pandoc upgrade: the first table added to a guide failed with
% "No counter 'none' defined" before this line existed.
%
% NB: pandoc parses this whole file as a template and knows nothing about LaTeX comments, so a
% literal dollar-if in the prose above would be read as a directive and fail the build. Keep
% template syntax out of the comments.
$common.latex()$

% After common.latex, so hyperref loads last as it expects to.
\usepackage[hidelinks]{hyperref}

\geometry{margin=2.4cm,top=2.0cm,bottom=2.2cm}

% No section numbers. Guides name their sections "Step 1 -- ...", and LaTeX's own numbering on top
% of that produces "2 Step 1 -- ...", which is worse than useless in a document people follow in
% order while typing.
\setcounter{secnumdepth}{0}

% Headings in the same sans as the title block, and unindented paragraphs separated by space
% rather than by indentation -- this is a set of instructions to scan, not prose to read.
\titleformat{\section}{\sffamily\bfseries\Large}{}{0pt}{}
\titleformat{\subsection}{\sffamily\bfseries\large}{}{0pt}{}
\titleformat{\subsubsection}{\sffamily\bfseries\normalsize}{}{0pt}{}
\titlespacing*{\section}{0pt}{18pt}{6pt}
\titlespacing*{\subsection}{0pt}{13pt}{4pt}
\titlespacing*{\subsubsection}{0pt}{11pt}{3pt}

\setlength{\parindent}{0pt}
\setlength{\parskip}{6pt}

% TeX Gyre and Latin Modern ship with TeX Live everywhere, so these need no font installation and
% no per-platform substitution.
%
% Addressed by FILE name rather than family name ("TeX Gyre Pagella") on purpose: TeX Live does not
% register its fonts with the operating system, so on a machine whose fontconfig/CoreText has never
% heard of them -- macOS, out of the box -- a family-name lookup fails and the build dies at
% \setmainfont. Filenames resolve through kpathsea, which always knows where TeX Live put them.
\setmainfont{texgyrepagella}[
  Extension      = .otf,
  UprightFont    = *-regular,
  BoldFont       = *-bold,
  ItalicFont     = *-italic,
  BoldItalicFont = *-bolditalic,
]
\setsansfont{texgyreheros}[
  Extension      = .otf,
  UprightFont    = *-regular,
  BoldFont       = *-bold,
  ItalicFont     = *-italic,
  BoldItalicFont = *-bolditalic,
]
% Latin Modern splits mono across two file stems: the roman/italic are lmmono10, the bold and
% oblique only exist under lmmonolt10 ("light"), so they cannot be reached with a single wildcard.
\setmonofont{lmmono10}[
  Extension        = .otf,
  UprightFont      = *-regular,
  ItalicFont       = *-italic,
  BoldFont         = lmmonolt10-bold,
  BoldItalicFont   = lmmonolt10-boldoblique,
  Scale            = MatchLowercase,
]

\definecolor{rule}{HTML}{C8C8C8}     % hairlines: the footer rule and the code-block frame
\definecolor{muted}{HTML}{555555}
\definecolor{shell}{HTML}{F4F4F4}

\setlist{noitemsep,topsep=4pt}

% --- institution marks -----------------------------------------------------------------------
% Logo files are dropped into logos/ by hand and are not in the repository (see logos/README.md),
% so this must degrade rather than fail: a missing file renders a framed placeholder naming the
% file that is wanted, which is visible in the PDF and obvious in review. Guides therefore build
% correctly from a fresh clone and simply get better as logos land.
\newcommand{\logodir}{$if(logodir)$$logodir$$else$logos$endif$}
\newcommand{\logoheight}{2.1ex}

\newcommand{\logoimg}[1]{\raisebox{-0.35ex}{\includegraphics[height=\logoheight]{#1}}}

% Without a logo file, fall back to the traditional footnote symbols in their conventional order,
% one per institution -- the same marks a paper would use. Each institution is registered once,
% in the order it appears in tutorial.json, so the symbol is a property of the institution and
% every person affiliated with it gets the same one.
\newcounter{institution}
\newcommand{\registerinstitution}[1]{%
  \stepcounter{institution}%
  \expandafter\xdef\csname instnum@#1\endcsname{\number\value{institution}}%
}
% Text-mode asterisks, not math mode: pandoc parses this file as a template, so a dollar-delimited
% math group anywhere in it is read as a directive and fails the build.
\newcommand{\affilsymbol}[1]{%
  \ifcase#1\or *\or \textdagger\or \textdaggerdbl\or \S\or \P\or \textbardbl\or
  **\or \textdagger\textdagger\or \textdaggerdbl\textdaggerdbl\else ?\fi
}
\newcommand{\affilmark}[1]{%
  \IfFileExists{\logodir/#1.pdf}{\,\logoimg{\logodir/#1.pdf}}{%
    \IfFileExists{\logodir/#1.png}{\,\logoimg{\logodir/#1.png}}{%
      \ifcsname instnum@#1\endcsname
        \textsuperscript{\affilsymbol{\csname instnum@#1\endcsname}}%
      \else
        \textsuperscript{?}%
      \fi}}%
}

% Assign the symbols up front, in the preamble, because the author rows are typeset before the
% legend that lists the institutions and both have to agree on which mark belongs to whom.
$for(institutions)$\registerinstitution{$institutions.id$}$endfor$

% --- header block ----------------------------------------------------------------------------
% The rules above and below the header are one command, not two similar-looking literals, so they
% cannot drift apart in weight or colour the next time either end is adjusted.
\newcommand{\headerrule}{\rule{\textwidth}{2pt}\par}
\newcommand{\peoplerow}[1]{{\sffamily\small #1\par}}

% Colour in the footer must be set with \textcolor, or inside an explicit group -- never with a
% bare \color switch. A bare switch is not bracketed, so it survives the page shipout and the NEXT
% page's body text comes out in it: page 1 reads black (its body is typeset before its footer) and
% everything from page 2 on is grey. Same reason \footrule is wrapped in braces.
\fancypagestyle{guide}{%
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0.4pt}
  \renewcommand{\footrule}{{\color{rule}\hrule height 0.4pt}}
  \fancyfoot[L]{\textcolor{muted}{\footnotesize\sffamily $conference$ · $title$}}
  \fancyfoot[R]{\textcolor{muted}{\footnotesize\sffamily \thepage}}
}
\pagestyle{guide}

% Fenced code blocks as a soft-grey panel rather than bare verbatim: these guides are almost
% entirely commands to copy, so the blocks are the content and should read as such. This has to
% come after common.latex, which is where Shaded and Highlighting are defined -- and inside the
% same conditional, because common.latex only defines them for a document that actually contains
% highlighted code. Renewing them unconditionally breaks any guide made purely of prose or tables.
$if(highlighting-macros)$
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},fontsize=\small,breaklines,breakanywhere}
\renewenvironment{Shaded}
  {\begin{tcolorbox}[breakable,colback=shell,colframe=rule,boxrule=0.4pt,arc=2pt,
                     left=6pt,right=6pt,top=5pt,bottom=5pt,enhanced jigsaw]}
  {\end{tcolorbox}}
$endif$

$if(institutions)$
  \vspace{3pt}

  \headerrule
